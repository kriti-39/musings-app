// Paste this into Firebase Console → Storage → Rules tab

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /receipts/{studentId}/{fileName} {
      // Any logged-in user can read receipts (admin/teacher reviewing)
      allow read: if request.auth != null;
      // Student uploads their own, or any authenticated user (admin uploading on behalf)
      allow write: if request.auth != null
        && request.resource.size < 5 * 1024 * 1024  // max 5MB
        && request.resource.contentType.matches('image/.*');
    }
  }
}
