// Paste this into Firebase Console → Firestore → Rules tab

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAuth() {
      return request.auth != null;
    }
    function role() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role;
    }
    function isAdmin() { return isAuth() && role() == 'admin'; }
    function isTeacher() { return isAuth() && role() == 'teacher'; }
    function isAdminOrTeacher() { return isAdmin() || isTeacher(); }

    // Users — students can read their own, teacher/admin can read all
    match /users/{userId} {
      allow read: if isAuth() && (request.auth.uid == userId || isAdminOrTeacher());
      allow create: if isAdminOrTeacher();
      allow update: if isAuth() && (request.auth.uid == userId || isAdminOrTeacher());
      allow delete: if isAdmin();
    }

    // Classes — student sees only their own, teacher/admin see all
    match /classes/{classId} {
      allow read: if isAuth() && (isAdminOrTeacher() || resource.data.studentId == request.auth.uid);
      allow create: if isAuth() && (isAdminOrTeacher() || request.resource.data.studentId == request.auth.uid);
      allow update: if isAuth() && (isAdminOrTeacher() || resource.data.studentId == request.auth.uid);
      allow delete: if isAdminOrTeacher();
    }

    // Recurring schedules — teacher/admin only
    match /recurringSchedules/{id} {
      allow read: if isAuth() && (isAdminOrTeacher() || resource.data.studentId == request.auth.uid);
      allow write: if isAdminOrTeacher();
    }

    // Availability (blocked slots) — anyone logged in can read, only teacher/admin can write
    match /availability/{id} {
      allow read: if isAuth();
      allow write: if isAdminOrTeacher();
    }

    // Payments — student sees only their own, teacher/admin see all. Only admin can update.
    match /payments/{id} {
      allow read: if isAuth() && (isAdminOrTeacher() || resource.data.studentId == request.auth.uid);
      allow create: if isAuth() && (isAdminOrTeacher() || request.resource.data.studentId == request.auth.uid);
      allow update: if isAdminOrTeacher();
      allow delete: if isAdmin();
    }

    // Notifications — user sees only their own
    match /notifications/{id} {
      allow read: if isAuth() && (isAdminOrTeacher() || resource.data.userId == request.auth.uid);
      allow create: if isAuth();
      allow update: if isAuth() && resource.data.userId == request.auth.uid;
      allow delete: if isAdmin();
    }
  }
}
